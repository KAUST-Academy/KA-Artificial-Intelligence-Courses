\begin{itemize}
    \item Deterministic Policy Gradient (Actor-Critic)
\end{itemize}
\vspace{0.5em}
\textbf{Objective:}
\[
J(\mu_\theta) = \int_{\mathcal{S}} \rho^{\mu}(s)\, r(s, \mu_\theta(s))\, ds = \mathbb{E}_{s \sim \rho^\mu}\!\left[ r(s, \mu_\theta(s)) \right]
\]
\end{frame}

\begin{frame}{Background -- DPG}
\begin{itemize}
    \item Deterministic Policy Gradient (Actor-Critic)
\end{itemize}
\textbf{Objective:}
\[
J(\mu_\theta) = \int_{\mathcal{S}} \rho^{\mu}(s)\, r(s, \mu_\theta(s))\, ds = \mathbb{E}_{s \sim \rho^\mu}\!\left[ r(s, \mu_\theta(s)) \right]
\]
\textbf{Policy Gradient:}
\[
\begin{aligned}
\nabla_\theta J(\mu_\theta) &= \int_{\mathcal{S}} \rho^{\mu}(s)\, \nabla_\theta \mu_\theta(s)\, \nabla_a Q^{w}(s, a)\big|_{a=\mu_\theta(s)}\, ds \\[2pt]
&= \mathbb{E}_{s \sim \rho^\mu}\!\left[ \nabla_\theta \mu_\theta(s)\, \nabla_a Q^{w}(s, a)\big|_{a=\mu_\theta(s)} \right]
\end{aligned}
\]
\end{frame}

\begin{frame}{Background -- DPG}
\textbf{Deterministic Policy Gradient (Actor-Critic)}
\vspace{0.5em}

\textbf{Objective:}
\[
J(\mu_\theta) = \int_{\mathcal{S}} \rho^{\mu}(s)\, r(s, \mu_\theta(s))\, ds = \mathbb{E}_{s \sim \rho^\mu}\!\left[ \rbox{r(s, \mu_\theta(s))} \right]
\]
\textbf{Policy Gradient:}
\[
\begin{aligned}
\nabla_\theta J(\mu_\theta) &= \int_{\mathcal{S}} \rho^{\mu}(s)\, \nabla_\theta \mu_\theta(s)\, \nabla_a \rbox{Q^{w}(s, a)\big|_{a=\mu_\theta(s)}}\, ds \\[2pt]
&= \mathbb{E}_{s \sim \rho^\mu}\!\left[ \nabla_\theta \mu_\theta(s)\, \nabla_a Q^{w}(s, a)\big|_{a=\mu_\theta(s)} \right]
\end{aligned}
\]
\end{frame}
