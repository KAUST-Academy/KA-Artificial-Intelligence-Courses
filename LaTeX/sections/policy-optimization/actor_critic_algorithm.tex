\begin{frame}{Actor-Critic Algorithm}
\begin{columns}
\begin{column}{0.54\textwidth}
\begin{algorithm}[H]
\scriptsize
\caption{One-step Actor-Critic (online)}
Initialise actor $\theta$, critic $\phi$\;
\For{each episode}{
    observe initial state $s_0$\;
    \For{$t = 0, 1, 2, \dots$}{
        sample $a_t \sim \pi_\theta(\cdot|s_t)$\;
        take $a_t$, observe $r_t, s_{t+1}$\;
        $\delta_t \leftarrow r_t + \gamma V_\phi(s_{t+1}) - V_\phi(s_t)$\;
        $\theta \leftarrow \theta + \alpha_\theta\, \nabla_\theta \log \pi_\theta(a_t|s_t)\, \delta_t$\;
        $\phi \leftarrow \phi - \alpha_\phi\, \delta_t\, \nabla_\phi V_\phi(s_t)$\;
    }
}
\end{algorithm}
\end{column}
\begin{column}{0.43\textwidth}
\begin{itemize}
    \item \textbf{Actor} steps along the policy gradient, weighted by $\delta_t$
    \item \textbf{Critic} regresses $V_\phi(s_t)$ toward $r_t + \gamma V_\phi(s_{t+1})$
    \item The same $\delta_t$ drives \emph{both} updates --- no full trajectory needed
    \item Note the signs: the actor \emph{ascends} ($+$) to maximize return; the critic \emph{descends} ($-$) to minimize $\mathbb{E}[\delta_t^2]$
    \item This is the pure \textbf{1-step TD} form. It is complete and it works --- but the single $\delta_t$ is a crude advantage estimate, which is what we fix next
\end{itemize}
\end{column}
\end{columns}
\end{frame}
