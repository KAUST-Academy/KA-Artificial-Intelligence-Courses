\begin{frame}{Model-Based RL}
\begin{itemize}
    \item Key Idea:
    \begin{itemize}
        \item It would be useful if we could approximately simulate the world! i.e.
              if we could predict the consequences of our actions
    \end{itemize}
    \item Advantages:
    \begin{itemize}
        \item Can efficiently learn model by supervised learning method
        \item Can reason about model uncertainty
    \end{itemize}
    \item Disadvantages:
    \begin{itemize}
        \item First learn a model, then construct a value function. This means we
              will have two sources of approximation error
    \end{itemize}
\end{itemize}
\end{frame}
