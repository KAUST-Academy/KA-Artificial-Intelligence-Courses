\section{Trust Region Policy Optimization (TRPO)}
\begin{frame}{}
    \LARGE Policy Optimization: \textbf{TRPO}
\end{frame}

\begin{frame}{TRPO: KL Trust Region}
\begin{itemize}
    \item TRPO formalises the constraint: maximize the surrogate objective subject to a KL-divergence bound:
    $$\max_\theta\; \mathcal{L}(\theta) \quad \text{s.t.} \quad \mathbb{E}_t\!\left[ D_{KL}\!\left(\pi_{\theta_\text{old}}(\cdot|s_t) \,\|\, \pi_\theta(\cdot|s_t)\right) \right] \leq \delta$$
    \item This keeps the new policy inside a \textbf{trust region} in policy space, not parameter space
    \item Solved with conjugate gradient + backtracking line search (second-order method)
\end{itemize}

\begin{frame}{TRPO: Line Search \& Cost}
\begin{itemize}
    \item TRPO uses backtracking line search with exponential decay to find the largest step that satisfies the KL constraint
\end{itemize}

\centering

\begin{itemize}
    \item \textbf{Cost:} requires computing (or approximating) the Fisher information matrix --- expensive and hard to implement at scale. This motivates PPO.
\end{itemize}
\end{frame}
