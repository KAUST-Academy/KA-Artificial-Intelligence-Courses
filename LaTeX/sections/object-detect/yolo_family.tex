\begin{frame}[allowframebreaks]{YOLO Family}
    \textbf{YOLO: One Network, One Pass}

    \begin{itemize}
        \item \textbf{Motivation:} the region-proposal detectors above run a classifier once per proposal, which is too slow for real time. YOLO predicts every box and class in a single forward pass.
        \item \textbf{Impact:} became the standard starting point for real-time detection on video and embedded hardware.
    \end{itemize}

    \end{figure}
\end{frame}
