% --- Learning outcomes ---
\begin{frame}[allowframebreaks]{Learning Outcomes}
  By the end of this session, you should be able to:
  \vspace{0.6em}
  \begin{itemize}\itemsep0.42em
    \item[\textcolor{red}{$\blacktriangleright$}] Choose appropriate benchmarks for a given capability claim, and explain what each does and does not measure.
    \item[\textcolor{red}{$\blacktriangleright$}] Run and interpret a standard evaluation harness, distinguishing \textbf{multiple-choice scoring} from \textbf{generative} and \textbf{execution-based} scoring.
    \item[\textcolor{red}{$\blacktriangleright$}] Design an \textbf{LLM-as-a-judge} protocol and measure its biases against human labels.
    \item[\textcolor{red}{$\blacktriangleright$}] Evaluate a \textbf{RAG} system along faithfulness, answer relevance, and context quality, and separate retrieval failures from generation failures.
    \item[\textcolor{red}{$\blacktriangleright$}] Detect and avoid \textbf{benchmark contamination}, and design an evaluation whose hygiene you can defend.
    \item[\textcolor{red}{$\blacktriangleright$}] Report results with \textbf{statistical rigor}: standard errors, clustering, paired comparisons, and power analysis.
    \item[\textcolor{red}{$\blacktriangleright$}] Build a small domain-specific eval set with rubrics, and align a judge to a domain expert.
  \end{itemize}
\end{frame}
