\begin{frame}{Model Learning}
\begin{itemize}
    \item Goal: Estimate model $\mathcal{M}_\phi$ from experience $s_1, a_1, r_1, \cdots, s_T$
    \item This is a supervised learning problem
    \item Learning $s, a \rightarrow r$ is a regression problem
    \item Learning $s, a \rightarrow s'$ is a density estimation problem
    \item We pick a loss function like MSE, KL Divergence etc., and then
          optimize
\end{itemize}
\end{frame}
