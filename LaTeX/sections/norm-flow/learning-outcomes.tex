\begin{frame}[allowframebreaks]{Normalizing Flow Models: Objectives}
    \begin{enumerate}
        \setlength{\itemsep}{-0.1em}
        \item \textbf{Understand the concept of normalizing flows.}
        \begin{enumerate}
            \setlength{\itemsep}{-0.5em}
            \item Learn how normalizing flows transform simple probability distributions into complex ones using invertible mappings.
        \end{enumerate}
        \item \textbf{Explore the mathematical foundations.}
        \begin{enumerate}
            \setlength{\itemsep}{-0.5em}
            \item Study the change of variables formula in probability.
            \item Analyze the role of the Jacobian matrix and its determinant in density estimation.
        \end{enumerate}
        \item \textbf{Examine key architectures and training objectives.}
        \begin{enumerate}
            \setlength{\itemsep}{-0.5em}
            \item NICE (Non-linear Independent Components Estimation)
            \item RealNVP (Real-valued Non-Volume Preserving)
            \item Glow (Generative Flow with Invertible 1x1 Convolutions)
            \item Flow matching (continuous-time flows and the current state of the art)
        \end{enumerate}
        \item \textbf{Discuss practical applications and summarize key takeaways.}
        \begin{enumerate}
            \setlength{\itemsep}{-0.5em}
            \item Review applications in generative modeling and density estimation.
            \item Highlight the main insights and lessons from this topic.
        \end{enumerate}
    \end{enumerate}
\end{frame}

\begin{frame}[allowframebreaks]{\textcolor{green}{$\Rightarrow$} Learning Outcomes}
    By the end of this session, you will be able to:
    \begin{enumerate}
        \setlength{\itemsep}{-0.1em}
        \item Understand what Normalizing Flows are and why they matter
        \item Apply change of variables in probability
        \item Understand the role of Jacobians and determinants
        \item Know how flow-based models like NICE, RealNVP, and Glow work
        \item Explain how flow matching trains a continuous-time flow by regression, and where it is used today
        \item Evaluate benefits and limitations of normalizing flows
    \end{enumerate}
\end{frame}
