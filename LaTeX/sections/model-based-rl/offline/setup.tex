\begin{frame}{A Hidden Assumption}
\begin{itemize}
    \item Every algorithm in this course has shared one quiet privilege:
          \textbf{if the agent acts badly, it can simply try again}
    \item DQN, PPO, DDPG, SAC, MBPO: all interleave \emph{learning} with
          \emph{acting}
    \item Even model-based RL fixed its early mistakes by going back out and
          collecting more data
    \item What if there is no ``going back out''?
\end{itemize}
\end{frame}

\begin{itemize}
    \item Given: a \textbf{fixed} dataset $\mathcal{D} = \{(s_i, a_i, r_i, s'_i)\}$,
          collected by an unknown \emph{behavior policy} $\pi_\beta$
    \item Goal: the best possible policy, \textbf{without a single new interaction}
\end{itemize}
\end{frame}

% === Why bother =========================================================
\begin{frame}{Why Would We Tie Our Own Hands?}
\begin{itemize}
    \item Healthcare: we cannot experiment on patients
    \item Autonomous driving: we cannot crash in order to learn
    \item Recommenders, industrial control: years of logs, no appetite for live
          exploration
    \item The situation of modern machine learning: \textbf{mountains of logged
          data, zero permission to explore}
\end{itemize}
\end{frame}
