\begin{itemize}
    \item \textbf{Actor update} (policy gradient):
    $$\theta \leftarrow \theta + \alpha_\theta\, \nabla_\theta \log \pi_\theta(a_t|s_t)\, \delta_t$$
    \item \textbf{Critic update} (TD learning):
    $$\phi \leftarrow \phi - \alpha_\phi\, \delta_t\, \nabla_\phi V_\phi(s_t)$$
    \item where $\delta_t = r_t + \gamma V_\phi(s_{t+1}) - V_\phi(s_t)$
    \item Updates happen \textbf{at every timestep} --- no full trajectory needed
\end{itemize}
\end{column}
