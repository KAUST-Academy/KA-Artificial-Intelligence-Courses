\begin{frame}[allowframebreaks]{fastText – Key Ideas}
    \begin{itemize}
        \item Developed by Facebook AI (2016)
        \item Builds on Word2Vec's skip-gram model by incorporating character n-grams
        \item Useful for morphologically rich languages and rare words
        \item Better handling of OOV (out-of-vocabulary) words: an unseen word still has n-grams, so its vector is the sum of those
        \item The same name also covers a companion supervised text classifier from the same group
    \end{itemize}

\begin{frame}{fastText – Example}
    \begin{itemize}
        \item Word \texttt{playing}, first padded with boundary symbols: \texttt{<playing>}
        \item Character n-grams (n=3): \texttt{<pl}, \texttt{pla}, \texttt{lay}, \texttt{ayi}, \texttt{yin}, \texttt{ing}, \texttt{ng>}
        \item The whole word is kept as one extra sequence: \texttt{<playing>}
        \item Word vector = sum of n-gram vectors
        \item The boundary symbols separate affixes from the same letters elsewhere: \texttt{<her>} (the word \emph{her}) is a different n-gram from \texttt{her} inside \emph{where}
        \item In practice every n-gram with $3 \leq n \leq 6$ is used, not only $n = 3$
    \end{itemize}
\end{frame}
