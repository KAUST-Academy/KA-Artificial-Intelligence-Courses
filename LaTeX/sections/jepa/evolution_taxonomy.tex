\begin{frame}{Evolution of SSL: A Brief History}
  \begin{itemize}\itemsep0.28em
    \item \textbf{Early methods:} rotation prediction, jigsaw, colorization. Cheap supervision, limited semantic depth.
    \item \textbf{Joint-embedding SSL (contrastive and non-contrastive):} SimCLR, MoCo (negatives); BYOL, VICReg (no negatives; VICReg instead regularizes embedding variance and covariance). Align embeddings across views while avoiding collapse.
    \item \textbf{Masked modeling / generative SSL:} MAE, BEiT. Reconstruct tokens or pixels in observation space.
    \item \textbf{Latent-target masked prediction:} iBOT (2022), data2vec (2022). Mask the input and predict an EMA teacher's \emph{representations}; the direct precursor to JEPA.
    \item \textbf{Predictive in latent space (JEPA):} I-JEPA, V-JEPA. Predict \emph{representations} of held-out regions rather than raw observations, with an explicit predictor module.
  \end{itemize}
  \vspace{0.35em}
  {\scriptsize Cross-method survey of SSL for ViTs: \href{https://arxiv.org/abs/2408.17059}{arXiv:2408.17059}.}
\end{frame}

\begin{frame}{Three Architectural Families in Self-Supervised Learning}
  \footnotesize
  \begin{center}
    \renewcommand{\arraystretch}{1.25}
    \begin{tabular}{@{}p{2.35cm}p{4.35cm}p{3.55cm}@{}}
      \hline
      \textbf{Family} & \textbf{Core mechanism} & \textbf{Representative examples} \\
      \hline
      Joint-embedding \newline \textit{(invariant)} & Pull together compatible views; contrastive or regularized losses prevent collapse. & SimCLR, MoCo, BYOL, VICReg \\
      \hline
      Generative \newline \textit{(reconstruction)} & Fill in missing input in pixel or token space; model must explain fine observation detail. & MAE, BEiT, VAEs, GAN-style objectives \\
      \hline
      JEPA \newline \textit{(latent prediction)} & Predict the embedding of a target region from context embeddings; captures semantics without a full pixel likelihood. & I-JEPA, V-JEPA, V-JEPA~2, Point-JEPA, CNN-JEPA \\
      \hline
    \end{tabular}
  \end{center}
\end{frame}
