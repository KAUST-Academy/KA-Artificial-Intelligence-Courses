% === MuZero (single high-level slide) ===================================
\begin{frame}{Planning with a Learned Model: AlphaZero \& MuZero}
\begin{itemize}
    \item Both \emph{plan} by \textbf{tree search} (MCTS), the discrete-move
          cousin of our CEM planner: grow a tree of moves, search the promising
          branches
    \item \textbf{AlphaZero}: MCTS $+$ a \emph{known} model, the rules of
          Go / chess handed to it for free
    \item \textbf{MuZero}: \emph{learns} the model instead, and the clever part is
          \emph{what} it learns: a model that predicts only \textbf{reward,
          value, and the best move}, never the raw pixels (a model of \emph{what
          matters}, not what the world looks like)
    \item So the same search works \emph{without being told the rules}: one
          architecture mastered Go, chess, shogi, \emph{and} Atari
          (Schrittwieser et al., 2020)
\end{itemize}
\end{frame}

% === Dreamer pointer (demoted to one-line pointer) ======================
\begin{frame}{Learning a Policy in the Model: Dreamer}
\begin{itemize}
    \item The other family learns a policy entirely \emph{inside} a learned world
          model: ``imagined'' roll-outs, no real interaction per update
    \item Its model is trained to \textbf{reconstruct what the world looks like}
          (predict the next observation), the opposite choice from MuZero's
          ``predict only what matters''
    \item \textbf{Dreamer / DreamerV3}: one fixed configuration solves 150+ tasks
          across Atari, continuous control, and Minecraft, with no per-task tuning
\end{itemize}
\vspace{0.3em}
\begin{center}
\emph{Two opposite bets on what a model is \emph{for}: MuZero --- don't waste
capacity on pixels. Dreamer --- pixels are a free, task-agnostic training
signal. Both won.}
\end{center}
\end{frame}
