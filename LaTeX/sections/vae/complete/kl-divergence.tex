\subsection{KL Divergence}
\begin{frame}{}
	\LARGE VAE: \textbf{Understanding KL Divergence}
\end{frame}

\begin{frame}[allowframebreaks]{What Is KL Divergence?}
\textbf{Definition:} the Kullback-Leibler divergence measures how different a distribution $p$ is from a reference distribution $q$:
\[
D_{\mathrm{KL}}(p \,\|\, q) \;=\; \sum_x p(x) \log \frac{p(x)}{q(x)} \;=\; \mathbb{E}_{x \sim p}\!\left[ \log p(x) - \log q(x) \right]
\]

\textbf{Properties:}
\begin{itemize}
    \item $D_{\mathrm{KL}}(p \,\|\, q) \geq 0$, with equality exactly when $p = q$.
    \item \textbf{Asymmetric:} $D_{\mathrm{KL}}(p \,\|\, q) \neq D_{\mathrm{KL}}(q \,\|\, p)$ in general, so it is not a distance metric.
    \item \textbf{Intuition:} with $\log_2$, it is the expected number of \emph{extra bits} needed to encode samples from $p$ using a code designed for $q$.
\end{itemize}

\framebreak

\textbf{KL divergence vs.\ cross-entropy:} the two are directly related:
\[
\underbrace{H(p, q)}_{\text{cross-entropy}} \;=\; \underbrace{H(p)}_{\text{entropy of } p} \;+\; \underbrace{D_{\mathrm{KL}}(p \,\|\, q)}_{\text{KL divergence}}
\]
\begin{itemize}
    \item In \textbf{classification}, $p$ is the fixed label distribution (one-hot), so $H(p)$ is a constant: minimizing cross-entropy and minimizing KL are \emph{the same thing}. That is why classifiers use cross-entropy.
    \item In a \textbf{VAE}, both $q(z \mid x)$ and the comparison to the prior depend on the model, and we need the divergence itself as a regularizer, not a classification loss. So the KL term appears explicitly.
\end{itemize}

\framebreak

\textbf{KL numerically (Gaussians):} for a 1-D Gaussian against the standard normal prior there is a closed form:
\[
D_{\mathrm{KL}}\!\left( \mathcal{N}(\mu, \sigma^2) \,\|\, \mathcal{N}(0, 1) \right) = \tfrac{1}{2}\left( \mu^2 + \sigma^2 - 1 - \log \sigma^2 \right)
\]
\begin{center}
\small
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}ccl@{}}
\hline
$\mu$ & $\sigma$ & $D_{\mathrm{KL}}$ \\
\hline
0 & 1 & $0$ \hfill (matches the prior exactly) \\
2 & 1 & $2.0$ \hfill (penalizes drifting away) \\
0 & 0.1 & $\approx 1.81$ \hfill (penalizes collapsing to a point) \\
0 & 2 & $\approx 0.81$ \hfill (penalizes spreading too wide) \\
\hline
\end{tabular}
\end{center}
\vspace{0.3em}
\begin{itemize}
    \item Summed over dimensions this is exactly the VAE regularizer from the ELBO section: $-\tfrac{1}{2} \sum_i \left( 1 + \log \sigma_i^2 - \mu_i^2 - \sigma_i^2 \right)$.
    \item Note the two-sided pressure: drifting from the prior \emph{and} becoming deterministic ($\sigma \to 0$) are both punished.
\end{itemize}
\end{frame}
