% ---------------------------------------------------------------- Slide 16 (problem analogy, annotated)
\begin{frame}{The meta-RL problem: an analogy}
\begin{columns}[t]
\begin{column}{0.5\textwidth}
\centering ``Generic'' learning:
\[ \theta^\star = \arg\min_\theta \mathcal{L}(\theta, \mathcal{D}^{\text{tr}})
   = f_{\text{learn}}(\mathcal{D}^{\text{tr}}) \]
\vspace{0.1em}

\centering Reinforcement learning:
\[ \theta^\star = \arg\max_\theta \mathbb{E}_{\pi_\theta}\!\left[R(\tau)\right]
   = f_{\text{RL}}(\mathcal{M}) \]
{\scriptsize \empr{RL is itself a function}: an MDP
$\mathcal{M}=\{\mathcal{S},\mathcal{A},\mathcal{P},r\}$ goes \emph{in}, an optimal policy
comes \emph{out}.}
\end{column}
\begin{column}{0.5\textwidth}
\centering ``Generic'' meta-learning:
\[ \theta^\star = \arg\min_\theta \sum_{i} \mathcal{L}\!\left(\phi_i, \mathcal{D}^{\text{ts}}_i\right),
   \;\; \phi_i = f_\theta(\mathcal{D}^{\text{tr}}_i) \]
\vspace{0.1em}

\centering Meta-reinforcement learning:
\[ \theta^\star = \arg\max_\theta \sum_i \mathbb{E}_{\pi_{\phi_i}}\!\left[R(\tau)\right],
   \;\; \phi_i = f_\theta(\mathcal{M}_i) \]
{\scriptsize \empr{learn the adapter} $f_\theta$: it turns an MDP $\mathcal{M}_i$ into a
good policy, and we train it across \emph{many} MDPs.}
\end{column}
\end{columns}
\vfill
\begin{center}
{\footnotesize $f_\theta$ is the \empr{meta-learner}: it maps a \emph{task}
$\mathcal{M}_i \mapsto$ \emph{policy parameters} $\phi_i$; we train $\theta$ so it does this
well across many tasks.}\\[2pt]
{\scriptsize (The parallel: the dataset $\mathcal{D}^{\text{tr}}$ becomes an MDP
$\mathcal{M}$; ``adapt, then score'' is the same recipe --- the loss just becomes a return.)}
\end{center}
\end{frame}

% ---------------------------------------------------------------- Slide 17 (task distribution, annotated)
\begin{frame}{The meta-RL problem: a distribution of tasks}
\begin{columns}[c]
\begin{column}{0.48\textwidth}
\[ \theta^\star = \arg\max_\theta \sum_i \mathbb{E}_{\pi_{\phi_i}}\!\left[R(\tau)\right],
   \;\; \phi_i = f_\theta(\mathcal{M}_i) \]
\footnotesize
\textbf{assumption:} $\mathcal{M}_i \sim p(\mathcal{M})$ --- tasks are \empr{drawn from a
distribution}; train and test tasks must \emph{share} it, or there is nothing to
generalise from.
\vspace{0.5em}

\textbf{meta-test} (the real exam): drop the trained learner into a \emph{brand-new} task
$\mathcal{M}_{\text{test}} \sim p(\mathcal{M})$; it must adapt,
$\phi = f_\theta(\mathcal{M}_{\text{test}})$, from a little experience.
\vspace{0.9em}

\centering
\begin{tikzpicture}[inner sep=1pt, ar/.style={-{Stealth[length=1.8mm]}}]
    \node (set) {$\{\mathcal{M}_1, \dots, \mathcal{M}_n\}$};
    \node[below left=0.55cm and -0.4cm of set, font=\scriptsize\itshape] (lbl) {meta-training MDPs};
    \draw[ar] (lbl) -- (set.south west);
\end{tikzpicture}
\end{column}
\begin{column}{0.52\textwidth}
\centering
{\footnotesize here $p(\mathcal{M})$ $=$ a distribution over \emph{target running speeds}:}\\[0.5em]
\includegraphics[width=\linewidth]{images/meta-rl-open/figures/examples-cheetah.jpg}\\[0.4em]
{\scriptsize each task rewards a different velocity; a new task is just a new target speed.}
\end{column}
\end{columns}
\vfill
\begin{center}\footnotesize
That pins down the \textbf{problem}. Building the adapter $f_\theta$ now takes three steps:\;
\empr{(1)} which task am I in? \;\; \empr{(2)} how to implement it? \;\; \empr{(3)} why does it explore?
\end{center}
\end{frame}

\begin{frame}{Building the meta-learner (3/3): why it explores}
\begin{columns}[c]
\begin{column}{0.52\textwidth}
\footnotesize
The RNN is trained to maximise reward over the \textbf{whole meta-episode} --- many
episodes back-to-back, hidden state never reset:
\[ \theta^\star = \arg\max_\theta \mathbb{E}_{\pi_\theta}\!\Big[\,
   \underbrace{\textstyle\sum_{t=0}^{T} r(s_t,a_t)}_{\text{summed across \emph{all} episodes}}\,\Big] \]
So \empr{spending early steps exploring pays off}: what the agent learns now earns more
reward later in the \emph{same} meta-episode. Exploration becomes \emph{optimal behaviour},
not a bolt-on.
\end{column}
\begin{column}{0.48\textwidth}
\centering

\begin{frame}{Black-Box Meta-RL: Algorithm}
\textbf{Meta-Training}
\begin{enumerate}
    \item Sample task $\mathcal{T}_i$
    \item Roll-out policy $\pi(a \mid s, \mathcal{D}^{\text{tr}}_i)$ for $N$ episodes
          \hfill {\footnotesize(under dynamics $p_i(s'\!\mid s,a)$ and reward $r_i(s,a)$)}
    \item Store sequence in replay buffer for task $\mathcal{T}_i$
    \item Update policy to maximize discounted return for all tasks.
    \item \textbf{Repeat} (go back to step 1).
\end{enumerate}
\vspace{0.5em}
\textbf{Meta-Test Time}
\begin{enumerate}
    \item Sample new task $\mathcal{T}_j$
    \item Roll-out policy $\pi(a \mid s, \mathcal{D}^{\text{tr}}_j)$ for up to $N$ episodes
\end{enumerate}
\end{frame}

\begin{frame}{Architectures for meta-RL}
\begin{center}\footnotesize
All implement the same $f_\theta(\mathcal{M}_i)$: read a history of experience
$\rightarrow$ produce context $\phi_i$. They differ in \emph{how they read it}.
\end{center}
\vspace{0.4em}
\begin{columns}[t]
\begin{column}{0.32\textwidth}
\begin{tcolorbox}[colback=sysblue!35, colframe=sysblue!80!black, title=\footnotesize\textbf{Recurrent} (RL$^2$)]
\footnotesize
Roll history through an \textbf{RNN}; hidden state $h_i$ \emph{is} the context.
\vspace{0.3em}

\empy{$+$} simple, general\\
\empy{$-$} long-horizon memory is hard
\end{tcolorbox}
\end{column}
\begin{column}{0.32\textwidth}
\begin{tcolorbox}[colback=boxgreen!45, colframe=boxgreen!70!black, title=\footnotesize\textbf{Attentive} (SNAIL)]
\footnotesize
\textbf{Temporal convolutions} $+$ \textbf{attention} over the whole history.
\vspace{0.3em}

\empy{$+$} pinpoints relevant past steps\\
\empy{$-$} heavier, context-window bound
\end{tcolorbox}
\end{column}
\begin{column}{0.32\textwidth}
\begin{tcolorbox}[colback=boxtan!50, colframe=boxtan!80!black, title=\footnotesize\textbf{Context encoder} (PEARL)]
\footnotesize
Encode experience into a \textbf{latent} $z$ (\emph{permutation-invariant}); condition
the policy on $z$.
\vspace{0.3em}

\empy{$+$} off-policy, sample-efficient\\
\empy{$-$} decouples explore vs.\ solve
\end{tcolorbox}
\end{column}
\end{columns}
\vfill

\;\textbullet\; Rakelly et al. PEARL (Efficient Off-Policy Meta-RL via Probabilistic Context Variables). ICML 2019.}
\end{frame}
