% --- What are Foundation Models? (General Ideas) ---
\begin{frame}{What are Foundation Models?}
  \begin{itemize}
    \item Large-scale models trained on massive and diverse datasets.
    \item Provide generic representations and capabilities across task domains.
    \item Enable “few-shot”, “zero-shot”, or “transfer learning” to downstream tasks.
    \item Examples: BERT, GPT, CLIP, SAM, etc.
    \item Key characteristics:
    \begin{itemize}
      \item Scalability: Millions or billions of parameters.
      \item Versatility: Perform well across multiple tasks and domains.
      \item Generality: Learn broad, reusable features---not task-specific ones.
    \end{itemize}
    \item Foundation for the current wave of AI advances in vision, language, tabular, and beyond.
  \end{itemize}
\end{frame}

% --- Foundation Models: Impact ---
\begin{frame}{Why Do Foundation Models Matter?}
  \begin{itemize}
    \item Rapidly accelerate the development of AI tools and systems.
    \item Lower the barrier for domain experts to build capable AI-powered solutions.
    \item Enable new applications in medicine, science, robotics, business, and more.
    \item Raise new questions:
      \begin{itemize}
        \item Training data composition and biases
        \item Computational demands
        \item Evaluation, safety, and control
      \end{itemize}
  \end{itemize}
\end{frame}

% --- Detection: Grounding DINO ---
\begin{frame}{Detection Foundation Model: Grounding DINO}
  \begin{itemize}
    \item \textbf{Grounding DINO}: A foundation model for open-set object detection with language grounding.
    \item \textbf{Key Ideas:}
      \begin{itemize}
        \item Can detect arbitrary objects and ground them using natural language queries.
        \item Builds on transformer architectures and leverages vision-language pretraining.
        \item Adaptable to various detection tasks without retraining for individual categories.
      \end{itemize}
    \item \textbf{Applications:}
      \begin{itemize}
        \item Open-vocabulary detection
        \item Multi-modal retrieval
        \item Robotics
      \end{itemize}
    \item \textbf{Strengths:} Generalization to unseen categories, flexible language grounding.
  \end{itemize}
\end{frame}

\begin{frame}{Grounding DINO: How It Works}
  \begin{itemize}
    \item Uses a transformer-based backbone for both image and text encoding.
    \item Matches object proposals in an image with user-provided text prompts (e.g. "a red backpack").
    \item Produces bounding boxes, labels, and confidence scores for detected objects.
    \item Supports real-time and scalable inference.
  \end{itemize}
  \centering
  \includegraphics[width=0.8\textwidth]{assets/New-Lecture_Foundation_models/grounding_dino_example.png}
\end{frame}

% --- Segmentation: SAM (Segment Anything Model) ---
\begin{frame}{Segmentation: Segment Anything Model (SAM)}
  \begin{itemize}
    \item \textbf{SAM} (\textit{Segment Anything Model}): A general, promptable segmentation foundation model.
    \item Released by Meta AI.
    \item Can segment objects in images based on points, boxes, or language prompts.
    \item Trained on $>$1 billion masks across 11 million images.
    \item Robust to diverse domains, objects, and prompts.
  \end{itemize}
\end{frame}

\begin{frame}{How does SAM work?}
  \begin{itemize}
    \item Encodes images into embeddings using a transformer backbone.
    \item User or downstream model provides prompts (points, boxes, text).
    \item The model predicts segmentation masks accordingly.
    \item Produces high-quality object masks across a wide range of settings without the need for fine-tuning.
  \end{itemize}
  \centering
  \includegraphics[width=0.85\textwidth]{assets/New-Lecture_Foundation_models/sam_schema.png}
\end{frame}

\begin{frame}{SAM 2 and SAM 3: What's New?}
  \begin{itemize}
    \item \textbf{SAM 2:}
      \begin{itemize}
        \item Improved efficiency and scalability.
        \item Better support for rare categories and edge cases.
        \item Enhanced prompt types, e.g., polygons, scribbles.
      \end{itemize}
    \item \textbf{SAM 3:}
      \begin{itemize}
        \item Multi-modal support: integrates with video, 3D, and multi-image inputs.
        \item Self-adaptive to challenging real-world conditions; employs continuous learning.
        \item Improved generalization to unseen domains (e.g., medical, aerial).
      \end{itemize}
    \item \textbf{Applications:} Data labeling, robotic perception, augmented reality, biology, medicine.
  \end{itemize}
\end{frame}

% --- Depth Estimation: Depth Anything ---
\begin{frame}{Depth Estimation Foundation Model: Depth Anything}
  \begin{itemize}
    \item \textbf{Depth Anything}: Foundation model for single image depth estimation.
    \item Learns to predict scene depth from generic images across domains.
    \item Leverages foundation model pretraining for generalized depth perception.
    \item \textbf{Applications:}
      \begin{itemize}
        \item Robotics and autonomous driving
        \item Virtual/augmented reality
        \item 3D scene understanding and reconstruction
      \end{itemize}
    \item \textbf{Strength}: Handles a variety of scenes (indoor, outdoor, synthetic, etc.) without task-specific supervision.
  \end{itemize}
\end{frame}

\begin{frame}{How Depth Anything Works}
  \begin{itemize}
    \item Uses a transformer-based encoder pretrained on a large corpus of image-depth pairs (and synthetic data).
    \item Maps RGB images to dense per-pixel depth maps.
    \item Generalizes to diverse lighting, occlusions, and camera settings.
    \item Integrates with other modalities (segmentation, detection) for richer scene analysis.
  \end{itemize}
  \centering
  \includegraphics[width=0.8\textwidth]{assets/New-Lecture_Foundation_models/depth_anything.png}
\end{frame}

% --- Keypoint Detection: SuperGlue ---
\begin{frame}{Keypoint Detection with SuperGlue}
  \begin{itemize}
    \item \textbf{SuperGlue}: Foundation model for keypoint matching and feature correspondence.
    \item Especially relevant to geometric computer vision: visual SLAM, mapping, structure-from-motion.
    \item \textbf{How it works:}
      \begin{itemize}
        \item Combines graph neural networks and attention to find matches between sets of keypoints in different images.
        \item Generalizes across scenes and keypoint detector types.
      \end{itemize}
    \item \textbf{Applications:}
      \begin{itemize}
        \item Visual localization and mapping
        \item 3D reconstruction
        \item Object pose estimation
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{SuperGlue: Model Architecture}
  \begin{itemize}
    \item Input: Sets of keypoints and descriptors from two images.
    \item Performs context aggregation and attention-based matching.
    \item Solves the permutation matching problem robustly.
    \item Delivers accurate and reliable matches---crucial for downstream robotics, AR, and vision.
  \end{itemize}
  \centering
  \includegraphics[width=0.78\textwidth]{assets/New-Lecture_Foundation_models/superglue.png}
\end{frame}

% --- Tabular Foundation Models: TabPFN ---
\begin{frame}{Tabular Foundation Models: TabPFN}
  \begin{itemize}
    \item \textbf{TabPFN}: Tabular Pretrained Foundation Network.
    \item Trained to “meta-learn” classification and regression solutions for tabular datasets.
    \item One forward pass -- inference in $<$1 second.
    \item \textbf{Key abilities:}
      \begin{itemize}
        \item Zero-shot predictions for new, small tabular datasets.
        \item No need for tuning or complex feature engineering.
      \end{itemize}
    \item \textbf{Outperforms conventional AutoML, especially on small datasets.}
    \item \textbf{Applications:} Science data analysis, health records, finance, industry.
  \end{itemize}
\end{frame}

\begin{frame}{TabPFN: Architecture and Capabilities}
  \begin{itemize}
    \item Learns a prior over the “space” of possible tabular learning problems.
    \item Uses transformers to consume (feature, target) pairs.
    \item Provides robust uncertainty estimates and calibrated predictions.
    \item Democratizes AI for small-scale data problems.
  \end{itemize}
  \centering
  \includegraphics[width=0.75\textwidth]{assets/New-Lecture_Foundation_models/tabpfn.png}
\end{frame}

% --- Medical Foundation Models: Tribe v2 ---
\begin{frame}{Medical Foundation Models: Tribe v2 (Google)}
  \begin{itemize}
    \item \textbf{Tribe v2}: Large, multi-modal foundation model for medical imaging and report interpretation.
    \item Developed by Google.
    \item \textbf{Key Features:}
      \begin{itemize}
        \item Trained across multiple medical imaging modalities (X-ray, MRI, CT, etc) and textual reports.
        \item Capable of multimodal reasoning---can answer medical questions, perform diagnosis, and cross-reference findings.
        \item Generalizes across new imaging modalities and unseen diseases.
      \end{itemize}
    \item \textbf{Strengths:}
      \begin{itemize}
        \item Few-shot adaptation to hospitals and rare conditions.
        \item Real-world clinical utility.
      \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}{Tribe v2: Example Applications}
  \begin{itemize}
    \item Automated report generation for radiology scans.
    \item Suggesting relevant prior cases or diagnoses.
    \item Real-time triage and anomaly detection in clinical workflows.
    \item Reducing diagnostic bias and supporting medical practitioners.
  \end{itemize}
  \centering
  \includegraphics[width=0.75\textwidth]{assets/New-Lecture_Foundation_models/tribe_v2.png}
\end{frame}

% --- Video Foundation Model: InternVideo ---
\begin{frame}{Video Foundation Model: InternVideo}
  \begin{itemize}
    \item \textbf{InternVideo}: Large-scale pre-trained video foundation model.
    \item Developed to generalize across diverse video understanding tasks.
    \item \textbf{Capabilities:}
      \begin{itemize}
        \item Action recognition
        \item Temporal localization
        \item Captioning and multi-modal reasoning with video and text
        \item Video segmentation and tracking
      \end{itemize}
    \item Learns rich spatio-temporal features adaptable to unseen video domains.
  \end{itemize}
\end{frame}

\begin{frame}{InternVideo: How It Works}
  \begin{itemize}
    \item Hierarchical transformer architecture processes temporal video clips and text prompts jointly.
    \item Pretrained on millions of video-text pairs.
    \item Can flexibly adapt to many video-based tasks with little (or no) fine-tuning.
    \item \textbf{Applications:} Video search, video QA, content moderation, robotics.
  \end{itemize}
  \centering
  \includegraphics[width=0.8\textwidth]{assets/New-Lecture_Foundation_models/internvideo.png}
\end{frame}
