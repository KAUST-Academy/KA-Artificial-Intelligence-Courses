\begin{frame}{Background Summary}
\vspace{0.5em}
\begin{itemize}
    \setlength{\itemsep}{0.7em}
    \item \textbf{DPG} \rref{1}: an update rule for \emph{deterministic} actors, so we can learn a policy on a continuous action domain without ever computing $\arg\max_a Q$
    \\[0.2em] \textcolor{red}{$\Rightarrow$ model-free, actor-critic}
    \pause
    \item \textbf{DQN} \rref{3}: value functions with neural networks, made to converge by two tricks
    \begin{itemize}
        \item \textbf{Target network} --- a frozen copy supplies the TD target, so we are not chasing our own tail
        \item \textbf{Replay buffer} --- decorrelates consecutive samples \; \textcolor{red}{$\Rightarrow$ off-policy}
    \end{itemize}
    \pause
    \item \textbf{DDPG} \rref{2}: learn \emph{both} the actor and the critic of DPG with neural networks, using the DQN tricks to keep it stable
\end{itemize}
\end{frame}
