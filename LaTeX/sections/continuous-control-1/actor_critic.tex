\begin{frame}{Background -- DPG}
\begin{itemize}
    \item Deterministic Policy Gradient (Actor-Critic)
\end{itemize}
\textbf{Objective:}
\[
J(\pi_\theta) = \int_{\mathcal{S}} \rho^{\pi}(s)\, r(s, \pi_\theta(s))\, ds = \mathbb{E}_{s \sim \rho^\pi}\!\left[ r(s, \pi_\theta(s)) \right]
\]
\textbf{Policy Gradient:}
\[
\nabla_\theta J(\pi_\theta) = \int_{\mathcal{S}} \rho^{\pi}(s)\, \nabla_\theta \pi_\theta(s)\, \nabla_a Q^{w}(s, a)\big|_{a=\pi_\theta(s)}\, ds = \mathbb{E}_{s \sim \rho^\pi}\!\left[ \nabla_\theta \pi_\theta(s)\, \nabla_a Q^{w}(s, a)\big|_{a=\pi_\theta(s)} \right]
\]
\end{frame}

\begin{frame}{Background -- DPG}
\textbf{Deterministic Policy Gradient (Actor-Critic)}
\vspace{0.5em}

\textbf{Objective:}
\[
J(\pi_\theta) = \int_{\mathcal{S}} \rho^{\pi}(s)\, r(s, \pi_\theta(s))\, ds = \mathbb{E}_{s \sim \rho^\pi}\!\left[ \rbox{r(s, \pi_\theta(s))} \right]
\]
\textbf{Policy Gradient:}
\[
\nabla_\theta J(\pi_\theta) = \int_{\mathcal{S}} \rho^{\pi}(s)\, \nabla_\theta \pi_\theta(s)\, \nabla_a \rbox{Q^{w}(s, a)\big|_{a=\pi_\theta(s)}}\, ds = \mathbb{E}_{s \sim \rho^\pi}\!\left[ \nabla_\theta \pi_\theta(s)\, \nabla_a Q^{w}(s, a)\big|_{a=\pi_\theta(s)} \right]
\]
\end{frame}
