\begin{frame}{Prior Work: Soft Q-Learning}
\begin{itemize}
    \item \textbf{Soft Q-Learning} (Haarnoja et al., 2017)
    \begin{itemize}
        \item off-policy algorithms under MaxEnt RL objective
        \item Learns $Q^{*}$ directly
        \item Instead of taking max action, sample the action
        \item use approximate inference methods to sample
        \begin{itemize}
            \item Stein variational gradient descent
        \end{itemize}
        \item not true actor-critic
    \end{itemize}
\end{itemize}
\[
Q(S_t, A_t) \leftarrow Q(S_t, A_t) + \alpha\big( R_{t+1} + \gamma \max_{a \in \mathcal{A}} Q(S_{t+1}, a) - Q(S_t, A_t) \big)
\]
\[
Q_{\text{soft}}(\mathbf{s}_t, \mathbf{a}_t) \leftarrow r_t + \gamma\, \mathbb{E}_{\mathbf{s}_{t+1} \sim p_{\mathbf{s}}}\!\left[ V_{\text{soft}}(\mathbf{s}_{t+1}) \right], \quad \forall \mathbf{s}_t, \mathbf{a}_t
\]
\[
V_{\text{soft}}(\mathbf{s}_t) \leftarrow \alpha \log \int_{\mathcal{A}} \exp\!\left( \tfrac{1}{\alpha} Q_{\text{soft}}(\mathbf{s}_t, \mathbf{a}') \right) d\mathbf{a}',\quad \forall \mathbf{s}_t
\]
\end{frame}
