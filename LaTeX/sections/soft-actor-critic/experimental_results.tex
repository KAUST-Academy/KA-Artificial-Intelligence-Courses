\begin{frame}{Experimental Results: Hyperparameter Sensitivity}
\begin{figure}
    \centering
    \includegraphics[width=\textwidth,height=0.72\textheight,keepaspectratio]{images/soft-actor-critic/figures/sac-hyperparam.jpg}
\end{figure}
\begin{itemize}
    \item[] {\footnotesize \textbf{Reward scale} (left): performance is \emph{dramatically} sensitive --- too small $\to$ no exploration signal, too large $\to$ near-deterministic; the right scale is task-specific. \textbf{Target smoothing $\tau$} (right): mild sensitivity, the usual stability/speed trade-off.}
    \item[$\Rightarrow$] {\footnotesize Reward scale is just a hand-tuned proxy for the entropy temperature $\alpha$ --- \emph{this} is the problem automatic temperature tuning solves.}
\end{itemize}
