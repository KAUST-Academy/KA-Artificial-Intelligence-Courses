\section{Learning Outcomes}
\begin{frame}{Learning Outcomes}
\begin{itemize}
    \item Understand the basic concepts of Computer Vision and its real-world applications.
    \item Describe how images are represented and processed in a computer.
    \item Explain the fundamental building blocks of Convolutional Neural Networks (CNNs): convolution, kernels/filters, padding, strides, pooling, activation functions, and fully connected layers.
\end{itemize}
\end{frame}
