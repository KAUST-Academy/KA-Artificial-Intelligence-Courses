\subsection{YOLOv3}
\begin{frame}{}
    \LARGE Object Detection: \textbf{YOLOv3}
\end{frame}

\begin{frame}[allowframebreaks]{YOLOv3}
    \begin{figure}
        \centering
        \includegraphics[width=1.0\textwidth,height=0.85\textheight,keepaspectratio]{images/object-detect/yolo-v3.png}
        \caption*{\scriptsize Source: A. Kathuria, ``What's new in YOLO v3?'', Towards Data Science, 2018.}
    \end{figure}
\framebreak
    \begin{itemize}
        \item \textbf{YOLOv3 (2018):}
        \begin{itemize}
            \item Uses Darknet-53 backbone (deeper, more powerful CNN)
            \item Multi-scale detection via Feature Pyramid Networks (FPN)
            \item Achieves $\sim$33 AP (57.9 AP@50) on the COCO dataset
            \item Improved small-object detection compared to previous YOLO versions
        \end{itemize}
        \item \textbf{Limitations:}
        \begin{itemize}
            \item Still misses some overlapping objects
            \item Less accurate than two-stage detectors (e.g., Faster R-CNN)
        \end{itemize}
    \end{itemize}
\end{frame}
