\begin{frame}{Probability Matching}
\begin{itemize}
    \item Probability Matching selects action $a$ according to probability that $a$
          is the optimal action
    \[
        \pi(a_{t+1} \mid h_t) = \mathbb{P}_{\mathcal{D}_t \sim \mathbb{P}[\mathcal{R} \mid h_t]}\!\left[\mathbb{E}_{\mathcal{D}_t}[r \mid a_{t+1}] > \mathbb{E}_{\mathcal{D}_t}[r \mid a], \; \forall a \neq a_{t+1}\right]
    \]
    \item Probability matching is optimistic in the face of uncertainty
    \item Because uncertain actions have higher probability of being max
    \item Can be difficult to compute analytically from posterior
\end{itemize}
\end{frame}

\begin{frame}[shrink=20]{Thompson Sampling}
\begin{itemize}
    \item Thompson Sampling implements probability matching
    \begin{align*}
        \pi(a_{t+1} \mid h_t) &= \mathbb{P}_{\mathcal{D}_t \sim \mathbb{P}[\mathcal{R} \mid h_t]}\!\left[\mathbb{E}_{\mathcal{D}_t}[r \mid a_{t+1}] > \mathbb{E}_{\mathcal{D}_t}[r \mid a], \; \forall a \neq a_{t+1}\right] \\
        &= \mathbb{E}_{\mathcal{D}_t \sim \mathbb{P}[\mathcal{R} \mid h_t]}\!\left[\mathbf{1}_{a_{t+1} = \operatorname*{arg\,max}_{a \in \mathcal{A}} \mathbb{E}_{\mathcal{D}_t}[r \mid a]}\right]
    \end{align*}
    \item Use Bayes law to compute posterior distribution $\mathbb{P}[\mathcal{R} \mid h_t]$
    \item Sample a reward distribution $\mathcal{D}_t$ from posterior $\mathbb{P}[\mathcal{R} \mid h_t]$
    \item Estimate Action-Value function with sample $\mathcal{D}_t$ as $\hat{Q}_t(a) = \mathbb{E}_{\mathcal{D}_t}[r \mid a]$
    \item Select action maximizing value of sample
    \[
        a_{t+1} = \operatorname*{arg\,max}_{a \in \mathcal{A}} \hat{Q}_t(a)
    \]
    \item Thompson Sampling achieves Lai-Robbins lower bound!
    \item \href{https://github.com/TikhonJelvis/RL-book/tree/master/rl/chapter14/ts_gaussian.py}{Educational Code} for Thompson Sampling for Gaussian Distributions
    \item \href{https://github.com/TikhonJelvis/RL-book/tree/master/rl/chapter14/ts_bernoulli.py}{Educational Code} for Thompson Sampling for Bernoulli Distributions
\end{itemize}
\end{frame}
