\begin{frame}{Lineage: From GLIDE to Stable Diffusion}
  \begin{itemize}\itemsep0.3em
    \item \textbf{GLIDE} (Nichol et al., 2021): a 3.5B-parameter text-conditional diffusion model. It compared CLIP guidance against classifier-free guidance and found human raters preferred the latter; fine-tuning it gave text-driven inpainting.
    \item \textbf{DALL·E~2} (Ramesh et al., 2022): two stages, a \textbf{prior} mapping a caption to a CLIP image embedding and a \textbf{decoder} diffusing an image from it. The shared CLIP space is what buys image variations and zero-shot edits.
    \item \textbf{Imagen} (Saharia et al., 2022): a \textbf{frozen}, text-only language model (T5-XXL) as the text encoder. Its headline finding is that scaling the language model improves fidelity and image-text alignment far more than scaling the image diffusion model.
    \item \textbf{Stable Diffusion} (Rombach et al., 2022): all three of the above denoise at pixel resolution and stack upsamplers on top. SD instead moves the diffusion into a \textbf{VAE latent space}, which is what made an openly released model that runs on a single consumer GPU possible.
  \end{itemize}
  \srcnote{GLIDE \href{https://arxiv.org/abs/2112.10741}{arXiv:2112.10741}; DALL·E~2 \href{https://arxiv.org/abs/2204.06125}{arXiv:2204.06125}; Imagen \href{https://arxiv.org/abs/2205.11487}{arXiv:2205.11487}; LDM \href{https://arxiv.org/abs/2112.10752}{arXiv:2112.10752}.}
\end{frame}

\begin{frame}{Application: Stable Diffusion 2.0}
  \footnotesize
  \textbf{What changed vs.\ Stable Diffusion v1?}
  \begin{itemize}\itemsep0.3em
    \item \textbf{New text encoder:} OpenCLIP (LAION + Stability AI) replaces CLIP ViT-L/14, giving stronger language understanding and image quality.
    \item \textbf{Native resolutions:} text-to-image at \textbf{512$\times$512} and \textbf{768$\times$768}.
    \item \textbf{Upscaler diffusion:} $\times 4$ super-resolution (e.g.\ 128$\rightarrow$512).
    \item \textbf{depth2img:} depth-guided image-to-image; structure-preserving synthesis from inferred depth + text.
    \item \textbf{Updated inpainting:} text-guided inpainting fine-tuned on the Stable Diffusion 2.0 base model.
  \end{itemize}
  \srcnote{\href{https://stability.ai/news-updates/stable-diffusion-v2-release}{Lopez, ``Stable Diffusion 2.0 Release,'' Stability AI (Nov 2022).}}
\end{frame}

\begin{frame}{Application: SDXL}
  \begin{itemize}\itemsep0.3em
    \item \textbf{SDXL} (Podell et al., 2023): the scaled-up successor to SD 2; for a long time the most-used open text-to-image model.
    \item A 2.6B-parameter U-Net (about 3$\times$ the 860M of SD 1/2) with \textbf{two text encoders} used together (OpenCLIP ViT-bigG + CLIP ViT-L).
    \item Native \textbf{1024$\times$1024} generation, plus micro-conditioning on image size and crop so the model learns from mixed-resolution data without artifacts.
    \item Optional \textbf{refiner} model: a second latent-diffusion stage specialized in high-frequency detail.
  \end{itemize}
\end{frame}

\begin{frame}{Inside Stable Diffusion 3: MM-DiT and Rectified Flow}
  \begin{itemize}\itemsep0.28em
    \item SD 3 changes both the backbone and the training objective (Esser et al., 2024).
    \item \textbf{Backbone:} the U-Net is replaced by a \textbf{multimodal diffusion transformer (MM-DiT)}, following the DiT line (Peebles \& Xie, 2023): text and image tokens flow through a joint transformer with separate weights per modality.
    \item \textbf{Objective:} \textbf{rectified flow}, i.e.\ flow matching on straight noise-data paths, instead of the DDPM objective; the paper's ablations found it superior at scale. (Flow matching was covered earlier in the course.)
    \item \textbf{Text encoders:} two CLIP encoders plus \textbf{T5-XXL} for long, compositional prompts and better in-image text.
    \item \textbf{FLUX.1} from Black Forest Labs (2024), a lab founded by members of the team behind VQGAN, Latent Diffusion and Stable Diffusion, continues this rectified-flow-transformer lineage as the strongest open release of its generation.
  \end{itemize}
\end{frame}

\begin{frame}{Stable Diffusion 3.5: Model Family and Hardware Footprint}
    \centering
    \includegraphics[width=\linewidth,height=0.29\textheight,keepaspectratio]{images/stable-diffusion/benchmark_medium_crop.png}\\[0.2em]
    {\scriptsize Source: Stability AI, \href{https://stability.ai/news-updates/introducing-stable-diffusion-3-5}{``Introducing Stable Diffusion 3.5''} (Oct 2024).}
      \begin{itemize}\itemsep0.35em
        \item \textbf{Customizability:} Large (8.1B), Large Turbo (8.1B) and Medium (2.5B) models.
        \item \textbf{Efficient Performance:} Optimized to run on standard consumer hardware; Medium and Large Turbo require no specialized GPUs.
        \item \textbf{Highly Accessible:} Medium needs 9.9\,GB of VRAM excluding the text encoders, making advanced diffusion imaging feasible on most consumer GPUs.
      \end{itemize}
\end{frame}
