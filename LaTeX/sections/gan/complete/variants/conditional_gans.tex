\subsection{Conditional GANs}
\begin{frame}{}
    \LARGE GAN Variant: \\[1.5ex] \textbf{Conditional GANs}
\end{frame}

\begin{frame}[allowframebreaks]{Conditional GAN}
\begin{itemize}
    \item In addition to random noise, a condition is added to the generator input. 
    \item The discriminator also receives this condition.
\end{itemize}
\begin{figure}
    \centering

    \caption*{\scriptsize [Source: Busayor, ``The Story of DCGANs, WGAN and CGANs,'' Analytics Vidhya, 2021]}
\end{figure}

\framebreak
\begin{itemize}
    \item Discriminator Loss:
    \begin{align*}
    \mathcal{L}^{(D)} \left( \theta^{(G)}, \theta^{(D)} \right) =\ &
    - \mathbb{E}_{(x,y) \sim p_{data}} \left[ \log D(x|y) \right] \\
    & - \mathbb{E}_{z \sim p_z(z),\ y \sim p_{data}(y)} \left[ \log(1 - D(G(z|y)\,|\,y)) \right]
    \end{align*}
    \item Generator Loss:
    $$
    \mathcal{L}^{(G)} \left( \theta^{(G)}, \theta^{(D)} \right) = - \mathbb{E}_{z \sim p_z(z),\ y \sim p_{data}(y)} \left[ \log D(G(z|y)\,|\,y) \right]
    $$
\end{itemize}
