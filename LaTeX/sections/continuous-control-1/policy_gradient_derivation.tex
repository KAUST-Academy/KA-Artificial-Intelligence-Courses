\section{Deterministic Policy Gradients (DPG)}
\begin{frame}{}
    \LARGE Continuous Control: \textbf{Deterministic Policy Gradients}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{The Day 3--4 Update Rule Stops Working}
\vspace{0.6em}
This is how we have trained every actor so far:
\[
\nabla_\theta \mathcal{J}(\theta) \;=\; \mathbb{E}\!\left[\, \nabla_\theta \log \pi_\theta(a_t \mid s_t)\; \hat{A}_t \,\right]
\]
It works by changing \textbf{probabilities}: make good actions more likely, bad actions less likely.
\pause
\vspace{1.2em}

\textcolor{red}{\textbf{But a deterministic actor has no probabilities.}}
\vspace{0.5em}
\begin{itemize}
    \setlength{\itemsep}{0.45em}
    \item $\mu_\theta(s)$ returns \emph{one} action. It is already certain --- there is no ``make it more likely''.
    \item So $\log \pi_\theta(a_t|s_t)$ has nothing to compute, and the expectation has nothing to average over.
\end{itemize}
\vspace{0.8em}
\centering
We need a different way to answer one question: \textbf{which way should I change $\theta$?}
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{DPG: Let the Critic \emph{Point} \rref{1}}
\vspace{0.4em}
Feed the actor's action into the critic: $Q_\phi\big(s, \mu_\theta(s)\big)$. That number says how good the actor's choice was. \textbf{We want it bigger} --- so climb it.
\pause
\vspace{0.7em}

Two questions, one after the other:
\begin{enumerate}
    \setlength{\itemsep}{0.35em}
    \item Which way should the \textbf{action} move to raise the value? \hfill $\rightarrow\;\; \nabla_a Q_\phi(s,a)$
    \item Which way should the \textbf{weights} move to move the action that way? \hfill $\rightarrow\;\; \nabla_\theta \mu_\theta(s)$
\end{enumerate}
\pause
\vspace{0.6em}
Multiply them. That is the \textbf{deterministic policy gradient}:
\[
\nabla_\theta \mathcal{J}(\mu_\theta) \;=\; \mathbb{E}_{s \sim \mathcal{D}}\!\left[\;
\rbox{\nabla_\theta \mu_\theta(s)} \;\cdot\; \rbox{\nabla_a Q_\phi(s,a)\big|_{a=\mu_\theta(s)}}
\;\right]
\]
\vspace{0.2em}
\centering
\textbf{In code this is one line:} maximise $Q_\phi\big(s,\mu_\theta(s)\big)$ with $\phi$ held fixed. Autograd does both steps.
\end{frame}

% ---------------------------------------------------------------------------
\begin{frame}{Old Rule vs.\ New Rule}
\vspace{0.4em}
\textbf{Stochastic (Days 3--4):}
\[
\nabla_\theta \mathcal{J}(\pi_\theta) = \mathbb{E}_{s,\; \rbox{a \sim \pi_\theta}}\!\left[\, \nabla_\theta \log \pi_\theta(a \mid s)\; Q_\phi(s, a) \right]
\]
\textbf{Deterministic (today):}
\[
\nabla_\theta \mathcal{J}(\mu_\theta) = \mathbb{E}_{s \sim \rbox{\mathcal{D}}}\!\left[\, \nabla_\theta \mu_\theta(s)\; \nabla_a Q_\phi(s, a)\big|_{a=\mu_\theta(s)} \right]
\]
\vspace{0.4em}
\begin{itemize}
    \setlength{\itemsep}{0.5em}
    \item The critic used to give a \textbf{score} --- a number saying how hard to push. Now it gives a \textbf{direction} --- which way to move the action.
    \item The old rule needs actions drawn from the \emph{current} policy ($a \sim \pi_\theta$). \textbf{That is what forced it to be on-policy.}
\end{itemize}
\end{frame}
