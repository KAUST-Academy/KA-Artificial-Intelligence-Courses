\section{Motivation}
\begin{frame}{Motivation}
    \large Why this topic matters:
        \begin{itemize}
            \item RNNs are sequential and struggle with long-range dependencies.
            \item Attention lets a model focus on the most relevant parts of the input.
            \item Transformers replace recurrence with self-attention --- fully parallelizable.
            \item They are the backbone of modern NLP and, increasingly, computer vision.
            \item \textit{"Attention is all you need."} --- \href{https://arxiv.org/abs/1706.03762}{Vaswani et al., 2017}.
        \end{itemize}
\end{frame}
