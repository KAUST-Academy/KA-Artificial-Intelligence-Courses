\section{Background}
\begin{frame}{Background -- DPG}
In Q-learning, we find deterministic policy by
\[
\mu^{k+1}(s) = \rbox{\arg\max_{a}}\, Q^{\mu^{k}}(s, a)
\]
\end{frame}

\begin{frame}{Background -- DPG}
In Q-learning, we find deterministic policy by
\[
\mu^{k+1}(s) = \rbox{\arg\max_{a}}\, Q^{\mu^{k}}(s, a)
\]
\textcolor{red}{\textbf{Problem:}} In large discrete action space or continuous action space, we can't plug in every possible action to find the optimal action!
\end{frame}

\begin{frame}{Background -- DPG}
In Q-learning, we find deterministic policy by
\[
\mu^{k+1}(s) = \rbox{\arg\max_{a}}\, Q^{\mu^{k}}(s, a)
\]
\textcolor{red}{\textbf{Problem:}} In large discrete action space or continuous action space, we can't plug in every possible action to find the optimal action!

\textcolor{green!50!black}{\textbf{Solution:}} Learn a function approximator for $\arg\max$, via gradient descent
\[
\mu^{k+1}(s) = \pi_{\rbox{\theta}}(s)
\]
\end{frame}
