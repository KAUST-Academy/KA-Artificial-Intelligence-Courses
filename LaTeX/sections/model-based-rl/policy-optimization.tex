\begin{frame}{Model-Based Policy Optimization}
\begin{itemize}
    \item We can also use model-based RL for policy optimization
    \item Key Idea:
    \begin{itemize}
        \item Augment data with model-simulated roll-outs
    \end{itemize}
    \item Algorithm (\textbf{MBPO}; Janner et al., 2019):
    \begin{enumerate}
        \item Collect data using current policy $\pi_\theta$, add to $D_{\text{env}}$
        \item Update model $p_\phi(s' \mid s, a)$ using $D_{\text{env}}$
        \item Collect synthetic roll-outs using $\pi_\theta$ in model $p_\phi$ from states in $D_{\text{env}}$;
              add to $D_{\text{model}}$
        \item Update policy $\pi_\theta$ (and critic $Q$) using $D_{\text{model}}$
        \item Go to step 1
    \end{enumerate}
\end{itemize}
\end{frame}

\begin{frame}{Model-Based Policy Optimization}
\begin{itemize}
    \item We can also use model-based RL for policy optimization
    \item Key Idea:
    \begin{itemize}
        \item Augment data with model-simulated roll-outs
    \end{itemize}
    \item Algorithm (\textbf{MBPO}; Janner et al., 2019):
    \begin{enumerate}
        \item Collect data using current policy $\pi_\theta$, add to $D_{\text{env}}$
        \item Update model $p_\phi(s' \mid s, a)$ using $D_{\text{env}}$
        \item Collect synthetic roll-outs using $\pi_\theta$ in model $p_\phi$ from states in $D_{\text{env}}$;
              add to $D_{\text{model}}$
        \item Update policy $\pi_\theta$ (and critic $Q$) using $D_{\text{model}}$
        \item Go to step 1
    \end{enumerate}
    \item Compatible with variety of model-free RL methods
    \item MBPO's design: \emph{short, branched} roll-outs from \emph{real} states
          keep model error from compounding
\end{itemize}
\end{frame}
