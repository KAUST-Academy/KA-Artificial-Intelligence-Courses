\newcommand{\rbox}[1]{%
  {\setlength{\fboxsep}{1pt}\setlength{\fboxrule}{0.7pt}%

}

% --- Red annotation boxes over native pseudocode -------------------------
% The algorithm slides used to be screenshots of the paper with hand-placed
% rectangles at hard-coded image fractions. The listings are now typeset with
% algorithm2e and \hlon draws the red rectangle around a chosen line on chosen
% overlays. The "off" branch draws the same node with no stroke, so the listing
% never shifts by a pixel between overlays.
% The negative kerns cancel the node's inner xsep so a boxed pseudocode line
% still starts at exactly the same x as the unboxed lines around it.
\newcommand{\hlbox}[2]{%
  \kern-2pt\begin{tikzpicture}[baseline=(hlnode.base)]
    \node[inner xsep=2pt, inner ysep=1.5pt, outer sep=0pt,
          draw=#1, line width=0.8pt] (hlnode) {#2};
  \end{tikzpicture}\kern-2pt%
}
% \hlon{<overlay spec>}{content} --- e.g. \hlon{<2>}{$a = \mu_\theta(s)$}
\newcommand{\hlon}[2]{\alt#1{\hlbox{red}{#2}}{\hlbox{none}{#2}}}

% Clickable reference citation: [n] jumps to the matching bibliography entry.
% Kept in normal text colour (like the Day 1--4 footnote markers) rather than
% hyperref's default red, so it is not visually distracting.
\newcommand{\rref}[1]{\hyperlink{ref:#1}{\textcolor{black}{[#1]}}}
