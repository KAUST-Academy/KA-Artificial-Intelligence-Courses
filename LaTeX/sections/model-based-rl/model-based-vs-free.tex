\section{Model-Based vs Model-Free RL}
% Slide 3
\begin{frame}{Model-Based and Model-Free RL}
\begin{itemize}
    \item Model-Free RL
    \begin{itemize}
        \item No model
        \item Learn value function (and/or policy) from experience
    \end{itemize}
\end{itemize}
\end{frame}

% Slide 4
\begin{frame}{Model-Based and Model-Free RL}
\begin{itemize}
    \item Model-Free RL
    \begin{itemize}
        \item No model
        \item Learn value function (and/or policy) from experience
    \end{itemize}
    \item Model-Based RL
    \begin{itemize}
        \item Learn a model from experience
        \item Plan value function (and/or policy) from model
    \end{itemize}
\end{itemize}
\end{frame}
