\subsection{Padding}
\begin{frame}{}
    \LARGE CNN Hyperparameter: \textbf{Padding}
\end{frame}

\begin{frame}[allowframebreaks]{Padding}
\begin{itemize}
    \item Applying convolution like this shrinks the output and loses information at the borders.
    \item Sometimes this is not desirable.
    \item We can pad the border with zeros.
\end{itemize}
