\begin{itemize}
    \item Self-Attention Layer is Permutation equivariant
    \item It doesn’t care about the order of the inputs!
    \item \textbf{Problem:} How can we encode ordered sequences like language or spatially ordered image features?
\end{itemize}
    
\end{frame}

\begin{frame}{Positional encoding}
\begin{itemize}
    \item Concatenate/add special positional encoding $p_j$ to each input vector $x_j$
    \item We use a function pos: $\mathbb{N} \rightarrow \mathbb{R}^D$ to process the position $j$ of the vector into a $D$-dimensional vector
    \item So, $p_j = pos(j)$

\end{itemize}
