\begin{frame}{Automatic Temperature Tuning}
\small
\textbf{Constrained problem} --- maximise return subject to a minimum policy entropy $\bar{\mathcal{H}}$ (so we set a \emph{target entropy} instead of a temperature):
\[
\max_{\pi_{0:T}} \mathbb{E}_{\rho_\pi}\!\left[ \sum_{t} r(\mathbf{s}_t, \mathbf{a}_t) \right] \;\; \text{s.t.}\;\; \mathbb{E}_{(\mathbf{s}_t, \mathbf{a}_t) \sim \rho_\pi}\!\left[ -\log \pi_t(\mathbf{a}_t \mid \mathbf{s}_t) \right] \geq \bar{\mathcal{H}}\;\; \forall t
\]

\textbf{Dual.} Introduce a Lagrange multiplier $\alpha_t\!\ge\!0$ per constraint (Levine \rref{3}). Each inner problem is the usual MaxEnt objective with $\alpha_t$ as temperature; the multiplier itself minimises the dual:
\[
\alpha_t^{*} = \arg\min_{\alpha_t \geq 0}\; \mathbb{E}_{\mathbf{a}_t \sim \pi_t^{*}}\!\left[\, -\alpha_t\big(\log \pi_t^{*}(\mathbf{a}_t \mid \mathbf{s}_t) + \bar{\mathcal{H}}\big) \,\right]
\]

\textbf{In practice} --- one SGD step on a single scalar, jointly with the actor/critic updates:
\[
J(\alpha) = \mathbb{E}_{\mathbf{a}_t \sim \pi_\theta}\!\left[\, -\alpha\big(\log\pi_\theta(\mathbf{a}_t\mid\mathbf{s}_t) + \bar{\mathcal{H}}\big) \,\right],\quad
\alpha \leftarrow \alpha - \lambda_\alpha\,\hat\nabla_\alpha J(\alpha)
\]
\vspace{-0.4em}
\begin{itemize}
    \item[] {\footnotesize $\lambda_\alpha$ is just the learning rate for that one scalar --- same Adam step size as the actor and critic.}
\end{itemize}
\end{frame}
\begin{frame}{SAC v2: The Same Loop, One Extra Line}
\begin{center}
{\scriptsize\textbf{\textcolor{red}{Red}} $=$ everything SAC v2 adds to the algorithm you already saw. $\alpha$ is now a \emph{learned} scalar, not a hyperparameter.\\
$\lambda_Q,\lambda_\pi,\lambda_\alpha$ are \textbf{learning rates} (Adam step sizes) for the critic, the actor and the temperature --- all $3\times10^{-4}$ in the paper.}
\end{center}
\begin{center}
\resizebox{\textwidth}{!}{%
\begin{minipage}{1.06\textwidth}
\begin{algorithm}[H]
\scriptsize
\DontPrintSemicolon
\LinesNotNumbered
\setlength{\algomargin}{0pt}
Initialize actor $\theta$, critics $\phi_1,\phi_2$;\ \ targets $\phi^{-}_i\!\leftarrow\!\phi_i$;\ \ replay $\mathcal{D}\leftarrow\emptyset$;\ \ \textcolor{red}{initialize $\alpha$;\ \ set target entropy $\bar{\mathcal{H}} = -\dim(\mathcal{A})$}\;
\For{each iteration}{
  \For{each environment step}{
    $\mathbf{a}_t\sim\pi_\theta(\,\cdot\mid\mathbf{s}_t)$;\ \ $\mathbf{s}_{t+1}\sim p$;\ \
    $\mathcal{D}\leftarrow\mathcal{D}\cup\{(\mathbf{s}_t,\mathbf{a}_t,r_t,\mathbf{s}_{t+1})\}$\;
  }
  \For{each gradient step}{
    sample minibatch from $\mathcal{D}$;\ \ $\tilde{\mathbf{a}}_{t+1}\sim\pi_\theta(\,\cdot\mid\mathbf{s}_{t+1})$\;
    $y = r_t + \gamma\big( Q^{\min}_{\phi^{-}}(\mathbf{s}_{t+1},\tilde{\mathbf{a}}_{t+1}) - \alpha\log\pi_\theta(\tilde{\mathbf{a}}_{t+1}\mid\mathbf{s}_{t+1})\big)$\;
    $\phi_i \leftarrow \phi_i - \lambda_Q\,\hat\nabla_{\phi_i}\big[\tfrac12(Q_{\phi_i}(\mathbf{s}_t,\mathbf{a}_t)-y)^2\big]$,\ \ $i\in\{1,2\}$\;
    $\theta \leftarrow \theta - \lambda_\pi\,\hat\nabla_\theta J_\pi(\theta)$ \ \ ($J_\pi$ critic term uses $Q^{\min}_{\phi}$)\;
    \textcolor{red}{$\alpha \leftarrow \alpha - \lambda_\alpha\,\hat\nabla_\alpha\big[\,-\alpha\big(\log\pi_\theta(\mathbf{a}_t\mid\mathbf{s}_t)+\bar{\mathcal{H}}\big)\,\big]$}\;
    $\phi^{-}_i \leftarrow \tau\phi_i+(1-\tau)\phi^{-}_i$,\ \ $i\in\{1,2\}$\;
  }
}
\end{algorithm}
\end{minipage}%
}
\end{center}
\vspace{-0.2em}
{\footnotesize \textbf{Read the new line:} if the actor is more decisive than the target ($\log\pi_\theta > -\bar{\mathcal{H}}$), $\alpha$ is pushed \emph{up} and exploration is bought back; if it is too random, $\alpha$ falls. One scalar, one gradient step, and the last brittle hyperparameter is gone.}
\begin{flushleft}
    \tiny Haarnoja et al.\ 2018b, \url{https://arxiv.org/abs/1812.05905}
\end{flushleft}
\end{frame}
