\begin{frame}{Hands-On: Code and Exercise}
  \begin{itemize}\itemsep0.3em
    \item[\textcolor{red}{$\blacktriangleright$}] \textbf{Interactive article:} \href{https://worldmodels.github.io/}{worldmodels.github.io} (Ha \& Schmidhuber) lets you play with the VAE latents and the dream environment in the browser.
    \item[\textcolor{red}{$\blacktriangleright$}] \textbf{DreamerV3:} the first author's reference reimplementation at \href{https://github.com/danijar/dreamerv3}{danijar/dreamerv3}; small control tasks train on a single GPU.
    \item[\textcolor{red}{$\blacktriangleright$}] \textbf{LeWM:} code at \href{https://github.com/lucas-maes/le-wm}{lucas-maes/le-wm}; small enough to read end to end, and a good base for course projects.
    \item[\textcolor{red}{$\blacktriangleright$}] \textbf{V-JEPA 2:} checkpoints on Hugging Face (e.g.\ \texttt{facebook/vjepa2-vitl-fpc64-256}, as covered earlier) for the latent-planning side.
    \item[\textcolor{red}{$\blacktriangleright$}] \textbf{Suggested exercise:} train DreamerV3 on a DeepMind Control task and log imagined vs.\ real rollouts side by side; then run LeWM's latent planner on PushT and compare how each system fails as the horizon grows.
  \end{itemize}
\end{frame}
