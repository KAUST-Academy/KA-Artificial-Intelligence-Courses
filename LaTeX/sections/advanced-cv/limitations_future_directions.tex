\section{Limitations and Future Directions}
\begin{frame}{}
    \LARGE Vision Transformers: \textbf{Limitations and Future Directions}
\end{frame}

\begin{frame}{Limitations of Transformers in Computer Vision}
    \begin{itemize}
        \item \textbf{Data hungry}: Performance drops significantly on small datasets.
        \item \textbf{Computationally expensive}: Quadratic complexity of self-attention.
        \item \textbf{Lacks built-in locality bias}: Unlike CNNs; some variants (e.g., Swin Transformer) address this.
        \item \textbf{Requires large-scale pretraining}: Needs datasets like JFT-300M, ImageNet-22K for strong performance.
    \end{itemize}
\end{frame}

\begin{frame}{Future Directions}
    \begin{itemize}
        \item \textbf{Efficient transformers}: \href{https://arxiv.org/abs/2006.04768}{Linformer}, \href{https://arxiv.org/abs/2009.14794}{Performer}, \href{https://arxiv.org/abs/2004.05150}{Longformer}.
        \item \textbf{Hybrid models}: Combining CNNs and Transformers (e.g., \href{https://arxiv.org/abs/2104.06399}{CoaT}, \href{https://arxiv.org/abs/2103.15808}{CvT}).
        \item \textbf{Low-resource ViTs}: Data-efficient training (e.g., \href{https://arxiv.org/abs/2012.12877}{DeiT}).
        \item \textbf{Vision-language models}: \href{https://arxiv.org/abs/2103.00020}{CLIP}, \href{https://arxiv.org/abs/2204.14198}{Flamingo}, \href{https://arxiv.org/abs/2303.08774}{GPT-4V}.
        \item \textbf{Hardware acceleration}: \href{https://arxiv.org/abs/2205.03436}{Edge ViTs}, quantized ViTs for deployment.
    \end{itemize}
\end{frame}
