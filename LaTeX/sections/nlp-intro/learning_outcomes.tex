\section{Learning Outcomes}
\begin{frame}{Learning Outcomes}
    \begin{itemize}
        \setlength{\itemsep}{0.6em}
        \item \textbf{Define} Natural Language Processing (NLP) and describe its core tasks
        \item \textbf{Apply} classical text classification with Naive Bayes and logistic regression
        \item \textbf{Explain and construct} N-grams (unigram, bigram, trigram) using sequence notation
        \item \textbf{Compute} N-gram probabilities, count matrices, and sequence probabilities
        \item \textbf{Apply} start/end tokens and handle unknown words (OOV, UNK)
        \item \textbf{Smooth} N-gram estimates so unseen contexts receive non-zero probability
        \item \textbf{Evaluate} a language model with perplexity
        \item \textbf{Recognize} the limitations of N-gram models and future directions
    \end{itemize}
\end{frame}
